Overview

매칭 완료 화면 진입 시 드라이버 프로필 사진이 Pulse 애니메이션과 함께 테두리 링이 나타납니다.
① Pulse + Ring — 진입 시 1회 실행. 사진이 줄어드는 동작에 맞춰 링 두께와 opacity가 함께 반응
② 프로필 사진 확대 뷰 — 탭 시 화면 내 dim + 원형 확대

실행 조건
매칭 완료 화면 최초 진입 시에만 실행합니다. 백그라운드 → 포그라운드 복귀 시에는 실행하지 않습니다.
실제 앱에서는 매칭 API 응답 콜백 시점에 트리거하세요.

DriverAvatarRingView (Pulse + Ring 통합)

Pulse와 Ring을 하나의 컴포넌트로 통합합니다. 링은 ZStack 고정 크기 기준으로 scale과 완전히 독립됩니다.
Task + await 패턴 사용 (iOS 16+, Swift 5.9 이상 권장 / WWDC 2025 기준 현행) 

struct DriverAvatarRingView: View {
    let image: Image
    let size: CGFloat = 60
    var onTap: (() -> Void)?

    @State private var scale: CGFloat = 1.0
    @State private var ringStroke: CGFloat = 2
    @State private var ringOpacity: Double = 0

    private let gap: CGFloat = 2
    private let pulseDuration: Double = 0.6

    var body: some View {
        ZStack {
            // 링: scale 독립, 바깥으로만 확장
            Circle()
                .stroke(Color(hex: "B9DDFD"), lineWidth: ringStroke)
                .frame(
                    width: size + gap * 2 + ringStroke,
                    height: size + gap * 2 + ringStroke
                )
                .opacity(ringOpacity)

            // 아바타: scale 적용
            image
                .resizable()
                .scaledToFill()
                .frame(width: size, height: size)
                .scaleEffect(scale)
                .clipShape(Circle())
                .onTapGesture { onTap?() }
        }
        .frame(width: size + gap * 2 + 4, height: size + gap * 2 + 4)
    }

    /// 매칭 완료 진입 시 호칬 (백그라운드 복귀 시 호칬 금지)
    func playPulseWithRing() {
        // Task + @MainActor: DispatchQueue.main.asyncAfter 대체 (iOS 16+)
        // 븷 해제 시 자동 취소, Swift Concurrency 표준 패턴
        Task { @MainActor in
            let easing = Animation.timingCurve(0.45, 0, 0.55, 1, duration: pulseDuration)

            // fade in
            withAnimation(.easeIn(duration: 0.2)) { ringOpacity = 0.4 }
            try await Task.sleep(for: .seconds(0.2))

            // pulse 3회
            for _ in 0..<3 {
                // down: scale 줄고 + ring 두껬워지고 + opacity 피크
                withAnimation(easing) { scale = 0.90; ringStroke = 4; ringOpacity = 1.0 }
                try await Task.sleep(for: .seconds(pulseDuration))
                // up: 복귀
                withAnimation(easing) { scale = 1.0; ringStroke = 2; ringOpacity = 0.4 }
                try await Task.sleep(for: .seconds(pulseDuration))
            }

            // fade out
            withAnimation(.easeOut(duration: 0.2)) { ringOpacity = 0 }
            try await Task.sleep(for: .seconds(0.2))
            ringStroke = 2
        }
    }
}

struct DriverProfileView: View {
    let driverImage: Image
    @State private var showExpandedPhoto = false

    var body: some View {
        ZStack {
            ScrollView {
                DriverAvatarRingView(image: driverImage, onTap: {
                    withAnimation(.spring(response: 0.35, dampingFraction: 0.64)) {
                        showExpandedPhoto = true
                    }
                })
            }
            if showExpandedPhoto {
                // Dim: ignoresSafeArea(false) → 화면 내부에만 적용
                Color.black.opacity(0.75)
                    .ignoresSafeArea(false)
                    .onTapGesture {
                        withAnimation(.easeInOut(duration: 0.25)) { showExpandedPhoto = false }
                    }
                // 확대 사진: screenWidth - 20pt (좌우 각 10pt 여백)
                let expandedSize = UIScreen.main.bounds.width - 20
                driverImage.resizable().scaledToFill()
                    .frame(width: expandedSize, height: expandedSize)
                    .clipShape(Circle())
                    .transition(.scale(scale: 0.3).combined(with: .opacity))
                    .allowsHitTesting(false)
            }
        }
    }
}
DriverAvatarRingView (Pulse + Ring 통합)

Kotlin 사용. Animatable + LaunchedEffect + async/awaitAll 패턴으로 병렬 애니메이션을 동기화합니다. 2026년 기준 공식 권장 패턴입니다. 

@Composable
fun DriverAvatarRingView(
    painter: Painter,
    size: Dp = 60.dp,
    onTap: () -> Unit = {},
    // 매칭 완료 신규 진입 시 true / 백그라운드 복귀 시 false 유지
    isNewMatch: Boolean = false
) {
    val gap = 2.dp; val minStroke = 2.dp; val maxStroke = 4.dp
    val pulseDurationMs = 600
    val easing = CubicBezierEasing(0.45f, 0f, 0.55f, 1f)

    // Animatable로 각 속성 독립 제어
    val scale = remember { Animatable(1f) }
    val ringStrokeDp = remember { Animatable(minStroke.value) }
    val ringOpacity = remember { Animatable(0f) }

    LaunchedEffect(isNewMatch) {
        if (!isNewMatch) return@LaunchedEffect
        delay(600L)
        ringOpacity.animateTo(0.4f, tween(200))
        repeat(3) {
            // async/awaitAll 사용을 위해 coroutineScope 필요
            coroutineScope {
            // down: 병렬 실행
            listOf(
                async { scale.animateTo(0.90f, tween(pulseDurationMs, easing = easing)) },
                async { ringStrokeDp.animateTo(maxStroke.value, tween(pulseDurationMs, easing = easing)) },
                async { ringOpacity.animateTo(1.0f, tween(pulseDurationMs, easing = easing)) }
            ).awaitAll()
            // up: 병렬 실행
            listOf(
                async { scale.animateTo(1.0f, tween(pulseDurationMs, easing = easing)) },
                async { ringStrokeDp.animateTo(minStroke.value, tween(pulseDurationMs, easing = easing)) },
                async { ringOpacity.animateTo(0.4f, tween(pulseDurationMs, easing = easing)) }
            ).awaitAll()
            } // end coroutineScope
        }
        ringOpacity.animateTo(0f, tween(200))
        ringStrokeDp.snapTo(minStroke.value)
    }

    Box(
        contentAlignment = Alignment.Center,
        modifier = Modifier.size(size + gap * 2 + maxStroke * 2)
    ) {
        // 링: Canvas로 직접 그리기 (scale 완전 독립)
        Canvas(modifier = Modifier.matchParentSize()) {
            val strokePx = ringStrokeDp.value.dp.toPx()
            val avatarRadiusPx = size.toPx() / 2
            val gapPx = gap.toPx()
            // 바깥으로만 확장: 안쪽 edge = avatarRadius + gap 고정
            val ringRadius = avatarRadiusPx + gapPx + strokePx / 2
            drawCircle(
                color = Color(0xFFB9DDFD), radius = ringRadius,
                style = Stroke(width = strokePx), alpha = ringOpacity.value
            )
        }
        // 아바타: scale 적용, 원형 클리핑
        Image(
            painter = painter, contentDescription = "드라이버 프로필",
            contentScale = ContentScale.Crop,
            modifier = Modifier.size(size)
                .graphicsLayer(scaleX = scale.value, scaleY = scale.value)
                .clip(CircleShape)
                .clickable(interactionSource = remember { MutableInteractionSource() },
                    indication = null, onClick = onTap)
        )
    }
}

@Composable
fun DriverProfileScreen(painter: Painter) {
    var showExpanded by remember { mutableStateOf(false) }
    val expandScale by animateFloatAsState(
        targetValue = if (showExpanded) 1.0f else 0.3f,
        animationSpec = spring(dampingRatio = 0.64f, stiffness = Spring.StiffnessMediumLow),
        label = "expandScale"
    )
    val dimAlpha by animateFloatAsState(
        targetValue = if (showExpanded) 0.75f else 0f,
        animationSpec = tween(250, easing = FastOutSlowInEasing), label = "dimAlpha"
    )
    Box(modifier = Modifier.fillMaxSize()) {
        Column { DriverAvatarRingView(painter = painter, onTap = { showExpanded = true }) }
        if (showExpanded || expandScale > 0.01f) {
            Box(modifier = Modifier.matchParentSize()
                .background(Color.Black.copy(alpha = dimAlpha))
                .clickable(interactionSource = remember { MutableInteractionSource() },
                    indication = null) { showExpanded = false },
                contentAlignment = Alignment.Center
            ) {
                val expandedSize = LocalConfiguration.current.screenWidthDp.dp - 20.dp
                Image(painter = painter, contentDescription = null,
                    contentScale = ContentScale.Crop,
                    modifier = Modifier.size(expandedSize).clip(CircleShape)
                        .graphicsLayer(scaleX = expandScale, scaleY = expandScale,
                            alpha = expandScale.coerceIn(0f, 1f))
                        .clickable(enabled = false) {}
                )
            }
        }
    }
}
